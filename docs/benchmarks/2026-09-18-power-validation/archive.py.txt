from pathlib import Path
import gzip,hashlib,json,shutil,tarfile
source=Path('/tmp/uscreen-t388-measured')
out=Path('docs/benchmarks/2026-09-18-power-validation');out.mkdir()
for name in ['analysis.json','deployment.json','original-preferences.json','final-preferences.json','final-health.json','plan.json']:
 shutil.copy2(source/name,out/name)
with tarfile.open(out/'observations.tar.gz','w:gz') as tar:
 for folder in sorted(source.glob('[0-9]*')):
  tar.add(folder,arcname=folder.name)
with tarfile.open(out/'collector-sources.tar.gz','w:gz') as tar:
 for name in ['run-baseline.py','observe.py','summarize.py','workload.py','visibility.py','xvisibility.py']:
  tar.add(Path('scripts/benchmarks')/name,arcname=name)
with tarfile.open(out/'control-smoke.tar.gz','w:gz') as tar:
 tar.add('/tmp/uscreen-t388-smoke2',arcname='smoke-only-not-a-power-trial')
 root=Path('/tmp/uscreen-t388-control-current')
 for p in root.rglob('*'):
  if p.is_file() and 'build' not in p.parts and '.gradle' not in p.parts and p.name!='local.properties':tar.add(p,arcname='project/'+str(p.relative_to(root)))
shutil.copy2('/tmp/uscreen-t388-control.png',out/'control-image.png')
for stem,name in [('measure','measure.py.txt'),('analyze','analyze.py.txt'),('archive','archive.py.txt')]:
 shutil.copy2('/tmp/uscreen-t388-'+stem+'.py',out/name)
(out/'run.log.gz').write_bytes(gzip.compress(Path('/tmp/uscreen-t388-measure.log').read_bytes(),mtime=0))
for name in ['uscreen-open-workspace','uscreen-open-clippy','uscreen-open-android','uscreen-open-tools-post-codec','uscreen-t462-complexity']:
 p=Path('/tmp')/(name+'.log')
 if p.exists():(out/(name+'.log.gz')).write_bytes(gzip.compress(p.read_bytes(),mtime=0))
(out/'SHA256SUMS').write_text(''.join(hashlib.sha256(p.read_bytes()).hexdigest()+'  '+p.name+'\n' for p in sorted(out.iterdir()) if p.is_file() and p.name!='SHA256SUMS'))
print(out)
