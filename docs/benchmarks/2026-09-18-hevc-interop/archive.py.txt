from pathlib import Path
import json,hashlib,tarfile,gzip,shutil,subprocess,struct
root=Path('docs/benchmarks/2026-09-18-hevc-interop');root.mkdir()
fixtures=[('original-vaapi',Path('/tmp/uscreen-t400-fixtures/motion-hevc_vaapi.bin')),('original-x265',Path('/tmp/uscreen-t400-fixtures/motion-libx265.bin'))]
for prefix,source in [('vaapi','/tmp/uscreen-t422-cases'),('features','/tmp/uscreen-t422-features')]:
    fixtures += [(prefix+'-'+p.parent.name,p) for p in sorted(Path(source).glob('*/stream.bin'))]
rows=[];temp=Path('/tmp/uscreen-t422-software');temp.mkdir()
with tarfile.open(root/'fixtures.tar.xz','w:xz') as tar:
    for name,path in fixtures:
        data=path.read_bytes();width,height,fps,count=struct.unpack_from('>4I',data,8);mime_size=struct.unpack_from('>H',data,24)[0];pos=26+mime_size
        config_size=struct.unpack_from('>I',data,pos)[0];config=data[pos+4:pos+4+config_size];pos+=4+config_size
        elementary=temp/(name+'.hevc')
        with elementary.open('wb') as stream:
            for _ in range(count):
                size=struct.unpack_from('>I',data,pos)[0];pos+=4;stream.write(data[pos:pos+size]);pos+=size
        assert pos==len(data)
        cmd=['ffmpeg','-v','error','-i',str(elementary),'-pix_fmt','nv12','-f','framemd5','-']
        result=subprocess.run(cmd,capture_output=True,text=True,timeout=30)
        frames=len([s for s in result.stdout.splitlines() if s and not s.startswith('#')])
        row=dict(name=name,sha256=hashlib.sha256(data).hexdigest(),width=width,height=height,fps=fps,frames=count,configuration_hex=config.hex(),software_returncode=result.returncode,software_frames=frames,software_errors=result.stderr,software_framemd5_sha256=hashlib.sha256(result.stdout.encode()).hexdigest())
        rows.append(row);tar.add(path,arcname=name+'.bin')
        print(name,'software frames',frames,'rc',result.returncode,flush=True)
(root/'fixtures.json').write_text(json.dumps(rows,indent=2)+'\n')
