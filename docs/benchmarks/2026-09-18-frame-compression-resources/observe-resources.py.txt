#!/usr/bin/env python3
"""T419: read-only live-session observation, not a controlled codec comparison."""
import argparse
import hashlib
import json
from pathlib import Path
import subprocess
import time

ROLES = {
    'app': 'com.uscreen',
    'vendor_codec': 'media.unisoc.codec2',
    'codec': 'media.codec',
    'software_codec': 'media.swcodec',
    'surfaceflinger': 'surfaceflinger',
    'composer': 'android.hardware.graphics.composer@2.4-service',
    'allocator': 'android.hardware.graphics.allocator@4.0-service',
}


def capture(serial, command):
    result = subprocess.run(['adb', '-s', serial, 'shell', command],
                            capture_output=True, text=True, check=True, timeout=12)
    return result.stdout.strip()


def save(folder, name, value):
    (folder / name).write_text(json.dumps(value, indent=2) + '\n')


def foreground(serial):
    raw = capture(serial, 'dumpsys activity activities')
    rows = [line.strip() for line in raw.splitlines() if 'topResumedActivity=' in line]
    if len(rows) != 1 or 'com.uscreen/.MainActivity' not in rows[0]:
        raise RuntimeError('UScreen no longer foreground; observation stopped')
    return rows


def process_ids(serial):
    rows = capture(serial, 'ps -A -o PID,NAME').splitlines()[1:]
    names = {line.split()[1]: int(line.split()[0]) for line in rows}
    return {role: names[name] for role, name in ROLES.items()}


def parse_stat(raw, page_size):
    fields = raw.rsplit(') ', 1)[1].split()
    return dict(pid=int(raw.split(' ', 1)[0]), start_ticks=int(fields[19]),
                cpu_ticks=int(fields[11]) + int(fields[12]),
                rss_bytes=int(fields[21]) * page_size, threads=int(fields[17]))


def read_process(serial, role, pid, page_size):
    command = f'cat /proc/{pid}/stat'
    if role == 'app':
        command = 'run-as com.uscreen ' + command
    begin = time.monotonic_ns()
    raw = capture(serial, command)
    end = time.monotonic_ns()
    return dict(raw=raw, monotonic_ns=(begin + end) // 2,
                observation_ns=end - begin, **parse_stat(raw, page_size))


def memory(serial, folder, phase, pids):
    for role, pid in pids.items():
        raw = capture(serial, f'dumpsys meminfo --local {pid}')
        (folder / f'meminfo-{phase}-{role}.txt').write_text(raw + '\n')


def device_state(serial, folder, phase):
    for name, command in [('battery', 'dumpsys battery'),
                          ('thermal', 'dumpsys thermalservice'),
                          ('memory', 'cat /proc/meminfo')]:
        (folder / f'{name}-{phase}.txt').write_text(capture(serial, command) + '\n')


def identity(serial, folder):
    apk = capture(serial, 'pm path com.uscreen').removeprefix('package:')
    # The installed package path comes from Android, not user-controlled shell text.
    import shlex
    result = dict(apk_sha256=capture(serial, 'sha256sum ' + shlex.quote(apk)).split()[0],
                  fingerprint=capture(serial, 'getprop ro.build.fingerprint'),
                  soc=capture(serial, 'getprop ro.soc.model'),
                  cores=int(capture(serial, 'getconf _NPROCESSORS_ONLN')),
                  page_size=int(capture(serial, 'getconf PAGESIZE')),
                  clock_ticks=int(capture(serial, 'getconf CLK_TCK')),
                  utc=time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime()),
                  boundary='uncontrolled installed foreground session; no content or cadence assertion',
                  script_sha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest())
    save(folder, 'metadata.json', result)
    return result


def observations(serial, folder, pids, meta):
    start = time.monotonic()
    with (folder / 'samples.jsonl').open('w') as output:
        for number in range(13):
            time.sleep(max(0, start + number * 5 - time.monotonic()))
            begin = time.monotonic_ns()
            top = foreground(serial)
            processes = {role: read_process(serial, role, pid, meta['page_size'])
                         for role, pid in pids.items()}
            row = dict(number=number, foreground=top, processes=processes,
                       collection_ns=time.monotonic_ns() - begin)
            output.write(json.dumps(row) + '\n')
            output.flush()
    foreground(serial)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--serial', required=True)
    parser.add_argument('--output', required=True, type=Path)
    args = parser.parse_args()
    args.output.mkdir()
    foreground(args.serial)
    pids = process_ids(args.serial)
    meta = identity(args.serial, args.output)
    save(args.output, 'processes.json', pids)
    device_state(args.serial, args.output, 'before')
    memory(args.serial, args.output, 'before', pids)
    try:
        observations(args.serial, args.output, pids, meta)
    finally:
        memory(args.serial, args.output, 'after', pids)
        device_state(args.serial, args.output, 'after')
    print('Completed 13 read-only samples; no workload or settings changed.')


if __name__ == '__main__':
    main()
